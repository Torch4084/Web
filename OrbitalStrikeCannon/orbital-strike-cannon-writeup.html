02 //

Linearizing Non-Associative Algebra

phase / analysis

The problem stated: "associativity was disabled". The core update loop multiplies an octonion state `M_i` by an RNG-derived octonion `R_i` and a fixed constant `alpha`:

nxt = add(o_mul(o_mul(R_i, moons[-1]), alpha), beta)

Because octonions are non-associative, `(R * M) * A != R * (M * A)`. However, I realized octonion multiplication is bilinear. For any fixed octonion `O`, multiplying by it can be expressed as a linear transformation matrix over F_p. The state transition can thus be cleanly represented as the matrix multiplication `R_alpha * L_Ri` applied to the coordinate vector.

I augmented the 8-dimensional state vector with the scalar `x` and a constant `1` to handle the affine `+ beta` shift, resulting in a 10-dimensional state vector.

With the state perfectly linearized, I had 9 unknowns: the 8 coordinates of `moon0` and the scalar `x0`.

The telemetry from each satellite gives us linear equations. However, some satellites are "dead" and feed fake data. I built a 15x9 system of equations for each satellite. Using Gaussian Elimination, an inconsistent system indicates a fake satellite, while a consistent system instantly yields the exact `moon0` and `x0` values!

I extracted the LCG parameters `u` and `v` from the first three public beacons to derive the final key material.

#!/usr/bin/env python3
import sys
import json
import hashlib
from pwn import *
import cannon

P = cannon.P

def solve_linear_system(A, b, P):
    n = len(A)
    m = len(A[0])
    
    M = [row[:] + [b[i]] for i, row in enumerate(A)]
    
    pivot_row = 0
    for col in range(m):
        pivot = -1
        for i in range(pivot_row, n):
            if M[i][col] != 0:
                pivot = i
                break
        if pivot == -1:
            continue
        
        M[pivot_row], M[pivot] = M[pivot], M[pivot_row]
        
        inv_val = pow(M[pivot_row][col], P - 2, P)
        for j in range(col, m + 1):
            M[pivot_row][j] = (M[pivot_row][j] * inv_val) % P
            
        for i in range(n):
            if i != pivot_row and M[i][col] != 0:
                factor = M[i][col]
                for j in range(col, m + 1):
                    M[i][j] = (M[i][j] - factor * M[pivot_row][j]) % P
                    
        pivot_row += 1

    for i in range(pivot_row, n):
        if M[i][m] != 0:
            return None
            
    if pivot_row < m:
        return None
        
    x = [0] * m
    for i in range(m):
        x[i] = M[i][m]
    return x

def solve():
    r = remote('orbital-f982a7eb366b.inst.omnictf.com', 1337, ssl=True)
    r.recvuntil(b'Survive the non-associative recoil.\n')
    json_str = r.recvuntil(b'}', drop=False).decode()
    while True:
        try:
            public = json.loads(json_str)
            break
        except json.JSONDecodeError:
            json_str += r.recv(1).decode()

    alpha = public['alpha']
    beta = public['beta']
    outer_a = public['outer_a']
    rng_beacons = public['rng_beacons']
    satellites = public['satellites']

    s0, s1, s2 = rng_beacons[0], rng_beacons[1], rng_beacons[2]
    u = ((s2 - s1) * pow(s1 - s0, P - 2, P)) % P
    v = (s1 - u * s0) % P

    states = cannon.build_state_expressions(alpha, beta, outer_a, rng_beacons)

    moon0, x0 = None, None

    for sat in satellites:
        start, end, step = sat['arange']
        mask_offset = sat['mask_offset']
        basis_rows_list = sat['basis']
        bias = sat['bias']
        coords = sat['coords']
        
        A = []
        b = []
        
        for t in range(5):
            idx = start + step * t
            mask = rng_beacons[idx + mask_offset]
            inv_mask = pow(mask, P - 2, P)
            frows = cannon.feature_rows(states, idx)
            
            for j in range(3):
                basis_j = basis_rows_list[t][j] if isinstance(basis_rows_list[0][0], list) else basis_rows_list[j] # Wait, basis is list of lists
                # Ah, let's check basis shape.
                # In cannon.py: basis_rows = [rnd_vec(rng, 11) for _ in range(COORDS_PER_SAMPLE)] -> list of 3 vectors of len 11.
                # It is fixed per satellite! Wait, `sat['basis']` is list of 3 vectors.
                
                eq_A = [0] * 9
                eq_C = 0
                for k in range(11):
                    for m in range(9):
                        eq_A[m] = (eq_A[m] + basis_j[k] * frows[k][m]) % P
                    eq_C = (eq_C + basis_j[k] * frows[k][9]) % P
                    
                val = (coords[t][j] - bias[j]) * inv_mask % P
                val = (val - eq_C) % P
                
                A.append(eq_A)
                b.append(val)
                
        x = solve_linear_system(A, b, P)
        if x is not None:
            moon0 = x[:8]
            x0 = x[8]
            break

    if moon0 is None:
        print("Failed to find valid satellite")
        sys.exit(1)

    secret_material = cannon.vec_to_bytes(moon0 + [x0, u, v])
    key = hashlib.sha256(b"OSC-KEY|" + secret_material).digest()
    firing_code = hashlib.sha256(key + b"|fire").hexdigest()[:32]
    
    r.recvuntil(b'firing code:\n')
    r.sendline(firing_code.encode())
    print(r.recvall().decode())

if __name__ == '__main__':
    solve()
