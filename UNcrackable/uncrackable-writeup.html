Algorithm Identification & Optimization

phase / identification

The README hinted: "It was a direct submission to NIST’s SHA-3 competition and was presented among the first-round candidates in 2009."

Eventually, I realized it was MD6-256. MD6 was submitted to the SHA-3 competition by Ron Rivest but withdrew early due to speed and provability issues. Testing MD6 in pure Python yielded a pathetic 300 hashes per second, which would take years to test the 42 billion combinations.

I downloaded the original MIT MD6 reference implementation in C and wrote a custom brute-forcer. I loaded the wordlist into memory, generated the case permutations on the fly, and used OpenMP (`#pragma omp parallel for`) to thread the MD6 compression loop across all cores.

#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <ctype.h>
#include <stdint.h>
#include <omp.h>
#include "md6.h"
#include "md6_nist.h"

// target: fdec178d06a81b3b4ff199ca43082ab58b328cd799640ddac726b71fb0b3ebd0
unsigned char target[32] = {
    0xfd, 0xec, 0x17, 0x8d, 0x06, 0xa8, 0x1b, 0x3b, 
    0x4f, 0xf1, 0x99, 0xca, 0x43, 0x08, 0x2a, 0xb5, 
    0x8b, 0x32, 0x8c, 0xd7, 0x99, 0x64, 0x0d, 0xda, 
    0xc7, 0x26, 0xb7, 0x1f, 0xb0, 0xb3, 0xeb, 0xd0
};

char words[15000000][64];
int num_words = 0;

int found = 0;

void test_mutation(const char* m, int len) {
    if (found) return;
    unsigned char hashval[32];
    Hash(256, (const BitSequence*)m, len * 8, hashval);
    
    // Quick check first byte
    if (hashval[0] == target[0] && hashval[1] == target[1]) {
        if (memcmp(hashval, target, 32) == 0) {
            #pragma omp critical
            {
                printf("\n[+] MATCH FOUND! Password: %s\n", m);
                found = 1;
            }
        }
    }
}

int main(int argc, char** argv) {
    if (argc < 2) {
        printf("Usage: %s <wordlist>\n", argv[0]);
        return 1;
    }
    FILE* f = fopen(argv[1], "r");
    if (!f) return 1;
    
    char buf[256];
    while (fgets(buf, sizeof(buf), f)) {
        int len = strlen(buf);
        while (len > 0 && (buf[len-1] == '\n' || buf[len-1] == '\r')) len--;
        buf[len] = 0;
        if (len < 3 || len >= 60) continue;
        if (num_words >= 14999999) break;
        strcpy(words[num_words++], buf);
    }
    fclose(f);
    printf("Loaded %d words.\n", num_words);

    #pragma omp parallel for schedule(dynamic, 100)
    for (int w = 0; w < num_words; w++) {
        char pw[128];
        strcpy(pw, words[w]);
        int pw_len = strlen(pw);
        
        // Rule a: capitalize first
        if (isalpha(pw[0])) {
            pw[0] = toupper(pw[0]);
        }
        
        char toggled[128];
        char mutated[128];
        
        // Rule b: toggle one of last 3
        for (int i = pw_len - 3; i < pw_len; i++) {
            if (i < 0) continue;
            if (isalpha(pw[i])) {
                strcpy(toggled, pw);
                if (islower(toggled[i])) toggled[i] = toupper(toggled[i]);
                else toggled[i] = tolower(toggled[i]);
                
                // Rule c: append 3 digits
                for (int d = 0; d < 1000; d++) {
                    sprintf(mutated, "%s%03d", toggled, d);
                    test_mutation(mutated, pw_len + 3);
                }
            }
        }
        if (w % 100000 == 0) {
            #pragma omp critical
            {
                printf("Progress: %d / %d\n", w, num_words);
            }
        }
    }
    printf("Not found.\n");
    return 0;
}
